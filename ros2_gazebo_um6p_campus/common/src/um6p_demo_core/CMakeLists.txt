cmake_minimum_required(VERSION 3.16)
project(um6p_demo_core)
cmake_policy(SET CMP0100 NEW)
find_package(ament_cmake REQUIRED)
find_package(gz_sim_vendor REQUIRED)
find_package(gz-sim8 REQUIRED)
find_package(gz_plugin_vendor REQUIRED)
find_package(gz-plugin2 REQUIRED COMPONENTS register)
find_package(gz_gui_vendor REQUIRED)
find_package(gz-gui8 REQUIRED)
find_package(Qt5 REQUIRED COMPONENTS Core Quick QuickControls2)
set(CMAKE_AUTOMOC ON)
qt5_add_resources(panel_resources config/panel.qrc)
add_library(ClearancePanel SHARED src/ClearancePanel.cc include/ClearancePanel.hh ${panel_resources})
target_compile_features(ClearancePanel PUBLIC cxx_std_17)
target_include_directories(ClearancePanel PRIVATE include)
target_link_libraries(ClearancePanel gz-gui8::gz-gui8 gz-plugin2::gz-plugin2 Qt5::Core Qt5::Quick Qt5::QuickControls2)
install(TARGETS ClearancePanel DESTINATION lib)
add_library(CampusKinematics SHARED src/CampusKinematics.cc)
target_compile_features(CampusKinematics PUBLIC cxx_std_17)
target_include_directories(CampusKinematics PRIVATE include)
target_link_libraries(CampusKinematics gz-sim8::gz-sim8 gz-plugin2::gz-plugin2)
add_executable(motion_check tests/motion_check.cc)
target_compile_features(motion_check PUBLIC cxx_std_17)
target_include_directories(motion_check PRIVATE include)
install(TARGETS CampusKinematics DESTINATION lib)
install(TARGETS motion_check DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
add_executable(panel_check tests/panel_check.cc)
target_link_libraries(panel_check Qt5::Core Qt5::Quick)
install(TARGETS panel_check DESTINATION lib/${PROJECT_NAME})
ament_package()
